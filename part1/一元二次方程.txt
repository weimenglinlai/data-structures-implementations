print("请输入一元二次方程ax^2+bx+c=0的系数a,b,c：")
a = float(input())
b = float(input())
c = float(input())
if a == 0:
    print("系数输入不正确！")
else:
    delta = b ** 2 - 4 * a * c
    if delta >= 0:
        x1 = (-b + (delta ** 0.5)) / (2 * a)
        x2 = (-b - (delta ** 0.5)) / (2 * a)
        print(f"一元二次方程的解为x1={x1:.2f},x2={x2:.2f}")
    else:
        print("一元二次方程无解")

#单链表结点类
class Node(object):
    def __init__(self,item):    #构造结点
        self.data = item     #设置数据域位指定数据
        self.next = None        #设置指针域为空

#单链表类
class SLinkList(object):
    def __init__(self):         #构造单链表
        self.head = None        #设置头指针
        self.length = 0         #设置单链表结点数为0

    def is_empty(self):          # 判断单链表是否为空
        return self.head is None

    def add(self, num):          # 头插法增加结点
        new_node = Node(num)
        new_node.next = self.head
        self.head = new_node
        self.length += 1

    def append(self, num):       # 尾插法增加结点
        new_node = Node(num)
        if self.is_empty():
            self.head = new_node
        else:
            current = self.head
            while current.next:
                current = current.next
            current.next = new_node
        self.length += 1

    def insert(self, num, index):  # 按指定索引下标序号插入结点
        if index < 0 or index > self.length:
            print("插入位置非法")
            return
        if index == 0:
            self.add(num)
            return
        new_node = Node(num)
        current = self.head
        for _ in range(index - 1):
            current = current.next
        new_node.next = current.next
        current.next = new_node
        self.length += 1

    def travel(self):            # 遍历输出单链表结点数据值
        current = self.head
        while current:
            print(current.data, end=' ')
            current = current.next
        print()

    def delete(self, index):     # 按索引下标序号删除指定结点
        if index < 0 or index >= self.length:
            print("删除位置非法")
            return
        if index == 0:
            self.head = self.head.next
        else:
            current = self.head
            for _ in range(index - 1):
                current = current.next
            current.next = current.next.next
        self.length -= 1

    def isContain(self, num):    # 查找是否包含指定值
        current = self.head
        index = 0
        while current:
            if current.data == num:
                return index
            current = current.next
            index += 1
        return -1

    def searchByIndex(self, index):  # 根据索引下标序号查找结点
        if index < 0 or index >= self.length:
            print("查找位置非法")
            return None
        current = self.head
        for _ in range(index):
            current = current.next
        print(current.data)
        return current.data

    def Alert(self, index, num):  # 根据索引下标序号修改结点数据值
        if index < 0 or index >= self.length:
            print("修改位置非法")
            return
        current = self.head
        for _ in range(index):
            current = current.next
        current.data = num
